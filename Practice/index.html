    What that means for studying
    You won't need to remember that it's getWheelRotation(); the appendix has it. You will need to know that zooming with the wheel means a MouseWheelListener, registered with addMouseWheelListener, and that a negative rotation means the wheel went up. So drill whole programs, and use the reference the way you'll use the appendix: look things up, then decide where they go.

    Filling in a code-completion question

    1. Find the event. What should happen, and which component causes it? A click on a button, typing in a field, the mouse moving over a panel.
    2. Pick the listener. Button or Enter in a text field → ActionListener. Check box, radio button, combo box → ItemListener. List → ListSelectionListener. Text changes → DocumentListener. Mouse → MouseListener/MouseMotionListener/MouseWheelListener. Keys → KeyListener. Window → WindowListener.
    3. Register it on the source. source.addXxxListener(…): this, a handler object, or new Xxx() { … }.
    4. Write the exact method. public void, the exact name, the right event type. An interface needs all its methods; an adapter only the ones you use.
    5. Read, convert, update. Read with getText(), isSelected(), getSelectedItem(), getX()…; convert with Integer.parseInt or a cast; update with setText, setIcon, setBackground…
    6. Finish the job. Changed what a panel draws → repaint(). Added or removed components → revalidate() and repaint(). Inside an anonymous class the frame is MyFrame.this, not this.

    How the drills work

    • Each drill is a real course program with parts removed. Type into the numbered boxes; spaces and line breaks don't matter.
    • Check marks every box and explains each answer. A missing ; counts as wrong, as it would in the exam, and Check says so.
    • Where other code would also work, Check accepts it and shows what the original had. For bigger blocks it looks for the parts that matter.
    • Show answers fills in the original code. Scores stay in this browser, so you can see which lectures still need work.

    private class GenderHandler implements ItemListener {

    private class InterestHandler implements ItemListener {

        public void itemStateChanged(ItemEvent e) {

            JCheckBox selected =
                (JCheckBox) e.getSource();

            if (e.getStateChange() == ItemEvent.SELECTED) {
                statusLabel.setText(
                    selected.getText() + " selected"

            } else {
                statusLabel.setText(

    // Register a student
    private void submitForm() {

        if (nameField.getText().trim().isEmpty()
                || emailField.getText().trim().isEmpty()
                || phoneField.getText().trim().isEmpty()) {

            JOptionPane.showMessageDialog(
                this, "Please complete all required fields."
            );
            return;
        }

        String email = emailField.getText().trim();

        if (!email.contains("@") || !email.contains(".")) {
            JOptionPane.showMessageDialog(
                this, "Please enter a valid email address."
            );
            return;
        }

        String gender = "Not selected";

        if (maleButton.isSelected())
            gender = "Male";
        else if (femaleButton.isSelected())
            gender = "Female";

        // Collect selected interests
        ArrayList<String> interests = new ArrayList<>();

        if (programmingBox.isSelected())
            interests.add("Programming");

        if (sportsBox.isSelected())
            interests.add("Sports");

        if (readingBox.isSelected())
            interests.add("Reading");

        // Create a Student object
        Student student = new Student(
            nameField.getText().trim(),
            email,
            phoneField.getText().trim(),
            (String) departmentCombo.getSelectedItem(),
            gender,
            interests
        );

        // Add the student object to the ArrayList
        students.add(student);

        // Display the registered student's information
        summaryArea.setText(student.toString());

        // Update the number of registered students
        studentCountLabel.setText(
            "Registered Students: " + students.size()
        );

        statusLabel.setText(
            "Student registered successfully!"
        );
    }

    // Display all registered students
    private void viewAllStudents() {

        if (students.isEmpty()) {
            JOptionPane.showMessageDialog(
                this, "No students have been registered yet."
            );
            return;
        }

        String allStudents = "";

        for (int i = 0; i < students.size(); i++) {

            allStudents += "Student " + (i + 1) + "\n";
            allStudents += students.get(i).toString();
            allStudents += "\n";
        }

        summaryArea.setText(allStudents);

        statusLabel.setText(
            "Displaying all registered students."
        );
    }

    // Reset form fields (does not delete registered students)
    private void resetForm() {

        nameField.setText("");
        emailField.setText("");
        phoneField.setText("");

        departmentCombo.setSelectedIndex(0);

        genderGroup.clearSelection();

        programmingBox.setSelected(false);
        sportsBox.setSelected(false);
        readingBox.setSelected(false);

        summaryArea.setText("");

        statusLabel.setText("Form has been reset.");
    }

        new StudentRegistrationForm();
    }

    package StudentRegistration;

import java.util.ArrayList;

public class Student {

    private String id;
    private String name;
    private String email;
    private String phone;
    private String department;
    private String gender;
    private ArrayList<String> interests;

    // Original constructor (6 parameters)
    public Student(String name, String email, String phone,
                   String department, String gender,
                   ArrayList<String> interests) {

        this.name = name;
        this.email = email;
        this.phone = phone;
        this.department = department;
        this.gender = gender;

        // Store a copy of the interests
        this.interests = new ArrayList<>(interests);
    }

    // New constructor (7 parameters), including university ID
    public Student(String id, String name, String email,
                   String phone, String department,
                   String gender, ArrayList<String> interests) {

        this.id = id;
        this.name = name;
        this.email = email;
        this.phone = phone;
        this.department = department;
        this.gender = gender;

        // Store a copy of the interests
        this.interests = new ArrayList<>(interests);
    }

    // Getter methods

    public String getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public String getEmail() {
        return email;
    }

    public String getPhone() {
        return phone;
    }

    public String getDepartment() {
        return department;
    }

    public String getGender() {
        return gender;
    }

    public ArrayList<String> getInterests() {
        return new ArrayList<>(interests);
    }

    @Override
    public String toString() {

        return "University ID: " + id +
               "\nName: " + name +
               "\nEmail: " + email +
               "\nPhone: " + phone +
               "\nDepartment: " + department +
               "\nGender: " + gender +
               "\nInterests: " + interests +
               "\n----------------------------";
    }
}

    private void searchStudent() {

        String id = JOptionPane.showInputDialog(
            this,
            "Enter the university ID:"
        );

        if (id == null || id.trim().isEmpty()) {
            return;
        }

        id = id.trim();

        Student student = students.get(id);

        if (student == null) {

            JOptionPane.showMessageDialog(
                this,
                "No student found with ID: " + id
            );

            return;
        }

        // Populate the form with the retrieved information
        idField.setText(student.getId());
        nameField.setText(student.getName());
        emailField.setText(student.getEmail());
        phoneField.setText(student.getPhone());

        departmentCombo.setSelectedItem(student.getDepartment());

        if (student.getGender().equals("Male")) {
            maleButton.setSelected(true);
        } else {
            femaleButton.setSelected(true);
        }

        // Clear previous interest selections
        programmingCheck.setSelected(false);
        sportsCheck.setSelected(false);
        readingCheck.setSelected(false);
        volunteeringCheck.setSelected(false);

        // Restore the student's interests
        ArrayList<String> interests = student.getInterests();

        for (String interest : interests) {

            if (interest.equals("Programming")) {
                programmingCheck.setSelected(true);
            }

            else if (interest.equals("Sports")) {
                sportsCheck.setSelected(true);
            }

            else if (interest.equals("Reading")) {
                readingCheck.setSelected(true);
            }

            else if (interest.equals("Volunteering")) {
                volunteeringCheck.setSelected(true);
            }
        }

        // Prevent changing the key while updating
        idField.setEditable(false);

        // Enable Update and disable Register
        updateButton.setEnabled(true);
        registerButton.setEnabled(false);

        summaryArea.setText(
            "Student found. You can edit the information in the form.\n\n"
            + student
        );
